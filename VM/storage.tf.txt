terraform {
  required_version = ">= 1.6.0"

  required_providers {
    azurerm = {
      source  = "hashicorp/azurerm"
      version = "~> 4.0"
    }
    random = {
      source  = "hashicorp/random"
      version = "~> 3.6"
    }
  }
}

provider "azurerm" {
  features {}
  storage_use_azuread = true
  subscription_id     = var.subscription_id
}

data "azurerm_client_config" "current" {}

variable "subscription_id" {
  description = "Azure subscription ID"
  type        = string
  default     = "92ade028-994d-424c-a639-48e33b16e332"
}

variable "location" {
  description = "Azure region for resources"
  type        = string
  default     = "westus3"
}

variable "environment" {
  description = "Environment name"
  type        = string
  default     = "prod"
}

locals {
  name_prefix = "pndt-${var.environment}"
  common_tags = {
    Environment = var.environment
    Workload    = "private-network-deployment-testing"
    ManagedBy   = "terraform"
  }
  peered_vnet_resource_group = "rg-${local.name_prefix}-networking"
  peered_vnet_name           = "vnet-${local.name_prefix}-peer"
  peered_subnet_name         = "snet-${local.name_prefix}-peer"
}

# Data source for the VM's private VNet
data "azurerm_virtual_network" "private_vnet" {
  name                = "vnet-${local.name_prefix}-private"
  resource_group_name = "rg-${local.name_prefix}"
}

# Data source for peered VNet
data "azurerm_virtual_network" "peered_vnet" {
  name                = local.peered_vnet_name
  resource_group_name = local.peered_vnet_resource_group
}

# Data source for peered subnet
data "azurerm_subnet" "peered_subnet" {
  name                 = local.peered_subnet_name
  virtual_network_name = data.azurerm_virtual_network.peered_vnet.name
  resource_group_name  = local.peered_vnet_resource_group
}

resource "azurerm_resource_group" "storage_rg" {
  name     = "rg-${local.name_prefix}-storage"
  location = var.location
  tags     = local.common_tags
}

resource "random_string" "storage_suffix" {
  length  = 6
  special = false
  upper   = false
}

# Private Storage Account
resource "azurerm_storage_account" "private_storage" {
  name                     = "st${replace(local.name_prefix, "-", "")}private${random_string.storage_suffix.result}"
  resource_group_name      = azurerm_resource_group.storage_rg.name
  location                 = azurerm_resource_group.storage_rg.location
  account_tier             = "Standard"
  account_replication_type = "LRS"
  min_tls_version          = "TLS1_2"
  
  # Restrict to private endpoints only
  public_network_access_enabled = false
  
  tags = local.common_tags
}

# Grant the current caller Storage Blob Data Contributor on the storage account
resource "azurerm_role_assignment" "storage_blob_contributor" {
  scope                = azurerm_storage_account.private_storage.id
  role_definition_name = "Storage Blob Data Contributor"
  principal_id         = data.azurerm_client_config.current.object_id
}

# Private Endpoint for Blob Storage (in Peered VNet)
resource "azurerm_private_endpoint" "storage_blob_endpoint" {
  name                = "pe-${local.name_prefix}-blob"
  location            = azurerm_resource_group.storage_rg.location
  resource_group_name = azurerm_resource_group.storage_rg.name
  subnet_id           = data.azurerm_subnet.peered_subnet.id

  private_service_connection {
    name                           = "psc-blob"
    private_connection_resource_id = azurerm_storage_account.private_storage.id
    subresource_names              = ["blob"]
    is_manual_connection           = false
  }

  tags = local.common_tags
}

# Private DNS Zone for Storage Account
resource "azurerm_private_dns_zone" "storage" {
  name                = "privatelink.blob.core.windows.net"
  resource_group_name = azurerm_resource_group.storage_rg.name

  tags = local.common_tags
}

# Link Private DNS Zone to Private VNet (so the VM can resolve the storage endpoint)
resource "azurerm_private_dns_zone_virtual_network_link" "storage_private_vnet_link" {
  name                  = "storage-private-vnet-link"
  resource_group_name   = azurerm_resource_group.storage_rg.name
  private_dns_zone_name = azurerm_private_dns_zone.storage.name
  virtual_network_id    = data.azurerm_virtual_network.private_vnet.id
}

# Link Private DNS Zone to Peered VNet
resource "azurerm_private_dns_zone_virtual_network_link" "storage_peered_vnet_link" {
  name                  = "storage-peered-vnet-link"
  resource_group_name   = azurerm_resource_group.storage_rg.name
  private_dns_zone_name = azurerm_private_dns_zone.storage.name
  virtual_network_id    = data.azurerm_virtual_network.peered_vnet.id
}

# DNS Record for Private Endpoint
resource "azurerm_private_dns_a_record" "storage_a_record" {
  name                = azurerm_storage_account.private_storage.name
  zone_name           = azurerm_private_dns_zone.storage.name
  resource_group_name = azurerm_resource_group.storage_rg.name
  ttl                 = 300
  records             = [azurerm_private_endpoint.storage_blob_endpoint.private_service_connection[0].private_ip_address]
}

# Blob Container for testing
resource "azurerm_storage_container" "test_container" {
  name               = "test-container"
  storage_account_id = azurerm_storage_account.private_storage.id

  depends_on = [
    azurerm_role_assignment.storage_blob_contributor,
    azurerm_private_dns_zone_virtual_network_link.storage_private_vnet_link,
    azurerm_private_dns_zone_virtual_network_link.storage_peered_vnet_link,
  ]
}
